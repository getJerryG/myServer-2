import request from 'supertest';
import app from '../../server';
import mongoose from 'mongoose';
import dotenv from 'dotenv';
 
// 加载环境变量
dotenv.config();
 
/**
 * 测试工具类
 */
export class TestUtils {
    /**
     * 获取Supertest请求实例
     */
    static getRequest() {
        return request(app);
    }
 
    /**
     * 生成随机字符串
     * @param length 字符串长度
     */
    static generateRandomString(length = 8): string {
        const chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789';
        let result = '';
        for (let i = 0; i < length; i++) {
            result += chars.charAt(Math.floor(Math.random() * chars.length));
        }
        return result;
    }
 
    /**
     * 生成随机数字
     * @param min 最小值
     * @param max 最大值
     */
    static generateRandomNumber(min = 0, max = 100): number {
        return Math.floor(Math.random() * (max - min + 1)) + min;
    }
 
    /**
     * 生成随机用户数据
     */
    static generateRandomUserData() {
        return {
            nickname: `test_user_${this.generateRandomString(6)}`,
            openId: `openid_${this.generateRandomString(16)}`,
            avatar: `https://example.com/avatar/${this.generateRandomString(8)}.jpg`,
            gamename: `game_${this.generateRandomString(6)}`,
        };
    }
 
    /**
     * 生成随机战队数据
     */
    static generateRandomClanData() {
        return {
            name: `clan_${this.generateRandomString(6)}`,
            description: `测试战队_${this.generateRandomString(12)}`,
            tag: this.generateRandomString(3).toUpperCase(),
            avatar: `https://example.com/clan/${this.generateRandomString(8)}.jpg`,
        };
    }
 
    /**
     * 生成随机赛事数据
     */
    static generateRandomContestData() {
        return {
            name: `contest_${this.generateRandomString(6)}`,
            description: `测试赛事_${this.generateRandomString(12)}`,
            startDay: new Date().toISOString(),
            endDay: new Date(Date.now() + 7 * 24 * 60 * 60 * 1000).toISOString(),
            maxSignTeams: this.generateRandomNumber(10, 50),
            status: 1,
        };
    }
 
    /**
     * 连接数据库
     */
    static async connectDB() {
        if (mongoose.connection.readyState === 0) {
            try {
                await mongoose.connect(process.env.MONGODB_URL || 'mongodb://localhost:27017/werewolf');
                console.log('Database connected for testing');
            } catch (error) {
                console.error('Database connection failed for testing:', error);
                throw error;
            }
        }
    }
 
    /**
     * 断开数据库连接
     */
    static async disconnectDB() {
        if (mongoose.connection.readyState !== 0) {
            await mongoose.disconnect();
            console.log('Database disconnected after testing');
        }
    }
 
    /**
     * 清除所有数据库数据
     */
    static async clearDB() {
        const collections = mongoose.connection.collections;
        for (const key in collections) {
            const collection = collections[key];
            await collection.deleteMany({});
        }
        console.log('Database cleared after testing');
    }
 
    /**
     * 测试前后的处理
     */
    static setupTestHooks() {
        // 所有测试开始前执行
        beforeAll(async () => {
            await this.connectDB();
        });
 
        // 所有测试结束后执行
        afterAll(async () => {
            await this.clearDB();
            await this.disconnectDB();
        });

            // 可以在这里添加每个测试后的处理
        });
    }
}
